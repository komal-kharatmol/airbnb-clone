function Protected({ children }) {
  const { user, loading } = useAuth();
  const loc = useLocation();
  if (loading) return <p className="p-10 text-center text-gray-500">Loading...</p>;
  return user ? children : <Navigate to="/login" state={{ from: loc.pathname }} replace />;
}

export default function App() {
  return (
    <>
      <Navbar />
      <main className="mx-auto max-w-7xl px-4 py-6 sm:px-8">
        <Routes>
          
          <Route path="/wishlists" element={<Wishlists />} />
<Route path="/" element={<Home />} />

          <Route path="/signup" element={<Auth mode="signup" />} />
          <Route path="/trips" element={<Protected><MyTrips /></Protected>} />

      </main>

  );
}

import { useEffect, useMemo, useState } from 'react';
import { useNavigate, useSearchParams } from 'react-router-dom';
import { AlertCircle, SearchX, X } from 'lucide-react';
import { api } from '../api';
import CategoryBar from '../components/CategoryBar';
import FilterModal, { DEFAULT_FILTERS, activeFilterCount, buildQuery } from '../components/FilterModal';
import ListingCard, { ListingCardSkeleton } from '../components/ListingCard';

const shortDate = (s) => new Date(s + 'T00:00:00').toLocaleDateString('en-IN', { day: 'numeric', month: 'short' });

export default function Home() {
  const [params] = useSearchParams();
  const nav = useNavigate();

  const location = params.get('location') || '';
  const checkIn = params.get('checkIn') || '';
  const checkOut = params.get('checkOut') || '';
  const guests = params.get('guests') || '';

  const [category, setCategory] = useState('All');
  const [filters, setFilters] = useState(DEFAULT_FILTERS);
  const [showFilters, setShowFilters] = useState(false);
  const [listings, setListings] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState('');
  const [retry, setRetry] = useState(0);

  const base = useMemo(() => ({ category, location, checkIn, checkOut, guests }), [category, location, checkIn, checkOut, guests]);
  const query = useMemo(() => buildQuery({ ...base, filters }), [base, filters]);
  const filterCount = activeFilterCount(filters);
  const nights = checkIn && checkOut ? Math.round((new Date(checkOut) - new Date(checkIn)) / 864e5) : 0;
  const hasSearch = location || (checkIn && checkOut) || guests;

  useEffect(() => {
    let dead = false;
    setLoading(true);
    setError('');
    api('/listings?' + query)
      .then((r) => !dead && setListings(r))
      .catch((e) => !dead && setError(e.message || 'Something went wrong'))
      .finally(() => !dead && setLoading(false));
    return () => { dead = true; };
  }, [query, retry]);

  const clearAll = () => { setCategory('All'); setFilters(DEFAULT_FILTERS); nav('/'); };

  const chips = [
    location && { key: 'loc', label: location },
    checkIn && checkOut && { key: 'dates', label: `${shortDate(checkIn)} – ${shortDate(checkOut)}` },
    guests && { key: 'guests', label: `${guests} guest${guests > 1 ? 's' : ''}` },
  ].filter(Boolean);

  return (
    <div>
      <CategoryBar value={category} onChange={setCategory} filterCount={filterCount} onOpenFilters={() => setShowFilters(true)} />

      {/* search summary */}
      {(hasSearch || filterCount > 0) && (
        <div className="mb-6 mt-4 flex flex-wrap items-center gap-2">
          {chips.map((c) => (
            <span key={c.key} className="rounded-full border border-[#DDDDDD] px-4 py-1.5 text-sm font-medium">{c.label}</span>
          ))}
          <button onClick={clearAll} className="flex items-center gap-1 rounded-full px-3 py-1.5 text-sm font-semibold underline hover:bg-[#F7F7F7]">
            <X size={14} /> Clear all
          </button>
        </div>
      )}

      {!loading && !error && listings.length > 0 && (
        <p className="mb-4 mt-2 text-sm text-[#717171]">{listings.length} stay{listings.length > 1 ? 's' : ''}{category !== 'All' ? ` in ${category}` : ''}</p>
      )}

      {error ? (
        <div className="mx-auto mt-16 max-w-md text-center">
          <AlertCircle size={40} className="mx-auto text-[#C13515]" />
          <h2 className="mt-4 text-xl font-semibold">We couldn't load stays</h2>
          <p className="mt-2 text-[#717171]">{error}. Make sure the backend is running on port 5000.</p>
          <button onClick={() => setRetry((n) => n + 1)} className="mt-6 rounded-lg bg-[#222222] px-6 py-3 font-semibold text-white hover:bg-black">Try again</button>
        </div>
      ) : loading ? (
        <div className="mt-6 grid grid-cols-1 gap-x-6 gap-y-10 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 2xl:grid-cols-5">
          {Array.from({ length: 10 }).map((_, i) => <ListingCardSkeleton key={i} />)}
        </div>
      ) : listings.length === 0 ? (
        <div className="mx-auto mt-16 max-w-md text-center">
          <SearchX size={44} className="mx-auto text-[#717171]" />
          <h2 className="mt-4 text-xl font-semibold">No exact matches</h2>
          <p className="mt-2 text-[#717171]">Try changing or removing some of your filters, or search a different place.</p>
          <button onClick={clearAll} className="mt-6 rounded-lg border border-[#222222] px-6 py-3 font-semibold hover:bg-[#F7F7F7]">Remove all filters</button>
        </div>
      ) : (
        <div className="grid grid-cols-1 gap-x-6 gap-y-10 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 2xl:grid-cols-5">
          {listings.map((l) => <ListingCard key={l.id} l={l} nights={nights} />)}
        </div>
      )}

      <FilterModal open={showFilters} onClose={() => setShowFilters(false)} filters={filters} onApply={setFilters} base={base} />
    </div>
  );
}

import { useEffect, useState } from 'react';
import { useLocation, useNavigate, useParams, useSearchParams } from 'react-router-dom';
import {
  Bed, Briefcase, Car, Check, ChevronLeft, ChevronRight, Coffee, Dumbbell, Flame, Heart, LayoutGrid,
  MapPin, Mountain, Share, Snowflake, Sparkles, Star, Tv, Users, UtensilsCrossed, Waves, Wifi,

import { api, fdate, money, onImgError } from '../api';

const DAY = 864e5;
const addDays = (s, n) => new Date(new Date(s).getTime() + n * DAY).toISOString().slice(0, 10);

const iconFor = (a) => {
  const s = a.toLowerCase();
  if (s.includes('wifi')) return Wifi;
  if (/pool|beach|sea/.test(s)) return Waves;
  if (s.includes('kitchen')) return UtensilsCrossed;
  if (/fire|bonfire|heating/.test(s)) return Flame;
  if (/air/.test(s)) return Snowflake;
  if (/park|bike|pickup/.test(s)) return Car;
  if (s.includes('gym')) return Dumbbell;
  if (/breakfast|coffee/.test(s)) return Coffee;
  if (/view|mountain/.test(s)) return Mountain;
  if (s.includes('tv')) return Tv;
  if (s.includes('work')) return Briefcase;
  if (/spa|butler/.test(s)) return Sparkles;
  return Check;
};

/* bento layouts for 1..5 photos on a 4x2 grid */
const LAYOUTS = {
  1: ['col-span-4 row-span-2'],
  2: ['col-span-2 row-span-2', 'col-span-2 row-span-2'],
  3: ['col-span-2 row-span-2', 'col-span-2', 'col-span-2'],
  4: ['col-span-2 row-span-2', 'col-span-2', 'col-span-1', 'col-span-1'],
  5: ['col-span-2 row-span-2', '', '', '', ''],
};

const readWish = () => { try { return JSON.parse(localStorage.getItem('wishlist')) || []; } catch { return []; } };

/* ----------------------- full-screen photo tour ----------------------- */
function PhotoTour({ images, start, title, onClose }) {
  const [i, setI] = useState(start);
  const n = images.length;
  const go = (d) => setI((p) => (p + d + n) % n);

  useEffect(() => {
    const onKey = (e) => {
      if (e.key === 'Escape') onClose();
      if (e.key === 'ArrowLeft') go(-1);
      if (e.key === 'ArrowRight') go(1);
    };
    document.body.style.overflow = 'hidden';
    window.addEventListener('keydown', onKey);
    return () => { document.body.style.overflow = ''; window.removeEventListener('keydown', onKey); };
  }, []); // eslint-disable-line react-hooks/exhaustive-deps

  const arrow = 'absolute top-1/2 z-10 grid h-11 w-11 -translate-y-1/2 place-items-center rounded-full border border-[#DDDDDD] bg-white shadow-md hover:scale-105 hover:shadow-lg';

  return (
    <div className="fixed inset-0 z-[70] flex animate-fade flex-col bg-white" role="dialog" aria-modal="true" aria-label="Photo tour">
      <div className="flex h-16 shrink-0 items-center justify-between px-4 sm:px-8">
        <button onClick={onClose} aria-label="Close photo tour" className="grid h-10 w-10 place-items-center rounded-full hover:bg-[#F7F7F7]">
          <ChevronLeft size={20} />
        </button>
        <p className="max-w-[60%] truncate text-sm font-semibold">{title} <span className="ml-2 font-normal text-[#717171]">{i + 1} / {n}</span></p>
        <span className="w-10" />
      </div>

      <div className="relative flex min-h-0 flex-1 items-center justify-center px-4 sm:px-20">
        {n > 1 && <button aria-label="Previous photo" onClick={() => go(-1)} className={`${arrow} left-3 sm:left-6`}><ChevronLeft size={20} /></button>}
        <img key={i} src={images[i]} onError={onImgError} alt={`${title} photo ${i + 1}`} className="max-h-full max-w-full animate-fade rounded-lg object-contain" />
        {n > 1 && <button aria-label="Next photo" onClick={() => go(1)} className={`${arrow} right-3 sm:right-6`}><ChevronRight size={20} /></button>}
      </div>

      <div className="no-scrollbar flex shrink-0 justify-center gap-2 overflow-x-auto p-4">
        {images.map((src, n2) => (
          <button key={n2} onClick={() => setI(n2)} aria-label={`Go to photo ${n2 + 1}`}
            className={`h-16 w-24 shrink-0 overflow-hidden rounded-lg border-2 ${n2 === i ? 'border-[#222222]' : 'border-transparent opacity-60 hover:opacity-100'}`}>
            <img src={src} onError={onImgError} alt="" className="h-full w-full object-cover" />
          </button>
        ))}
      </div>
    </div>
  );
}

export default function ListingDetail() {
  const { id } = useParams();
  const { user } = useAuth();
  const nav = useNavigate();
  const loc = useLocation();
  const [params] = useSearchParams();
  const [toast, showToast] = useToast();
  const today = new Date().toISOString().slice(0, 10);

  const [l, setL] = useState(null);
  const [error, setError] = useState('');
  const [tour, setTour] = useState(null); // photo index or null
  const [liked, setLiked] = useState(false);
  const [moreDesc, setMoreDesc] = useState(false);
  const [moreAmen, setMoreAmen] = useState(false);
  const [moreRev, setMoreRev] = useState(false);

  const qIn = params.get('checkIn') || '';
  const qOut = params.get('checkOut') || '';
  const [start, setStart] = useState(qIn >= today ? qIn : '');
  const [end, setEnd] = useState(qIn >= today && qOut > qIn ? qOut : '');
  const [guestCount, setGuestCount] = useState(1);
  const [msg, setMsg] = useState('');
  const [busy, setBusy] = useState(false);

  useEffect(() => {
    let dead = false;
    setL(null); setError('');
    api('/listings/' + id)
      .then((d) => {
        if (dead) return;
        setL(d);
        setLiked(readWish().includes(d.id));
        const g = parseInt(params.get('guests'), 10) || 1;
        setGuestCount(Math.min(Math.max(g, 1), d.guests));
      })
      .catch((e) => !dead && setError(e.message));
    window.scrollTo(0, 0);
    return () => { dead = true; };
  }, [id]); // eslint-disable-line react-hooks/exhaustive-deps

  if (error) {
    return (
      <div className="mx-auto mt-20 max-w-md text-center">
        <h1 className="text-2xl font-semibold">Listing unavailable</h1>
        <p className="mt-2 text-[#717171]">{error}</p>
        <button onClick={() => nav('/')} className="mt-6 rounded-lg bg-[#222222] px-6 py-3 font-semibold text-white hover:bg-black">Back to home</button>
      </div>
    );
  }
  if (!l) return <DetailSkeleton />;

  /* derived */
  const imgs = l.images;
  const n = Math.min(imgs.length, 5);
  const layout = LAYOUTS[n];
  const nights = start && end ? Math.round((new Date(end) - new Date(start)) / DAY) : 0;
  const invalid = !!(start && end && nights < 1);
  const clash = nights > 0 && l.bookings.some((b) => new Date(b.startDate) < new Date(end) && new Date(b.endDate) > new Date(start));
  const own = !!user && user.id === l.hostId;
  const total = nights * l.price;
  const canReserve = nights > 0 && !invalid && !clash && !own;

  const toggleLike = () => {
    const list = readWish();
    const next = list.includes(l.id) ? list.filter((x) => x !== l.id) : [...list, l.id];
    try { localStorage.setItem('wishlist', JSON.stringify(next)); } catch { /* ignore */ }
    setLiked(next.includes(l.id));
    showToast(next.includes(l.id) ? 'Saved to your wishlist' : 'Removed from your wishlist');
  };

  const share = async () => {
    try { await navigator.clipboard.writeText(window.location.href); showToast('Link copied to clipboard'); }
    catch { showToast('Copy the link from your address bar'); }
  };

  const reserve = async () => {
    if (!user) return nav('/login', { state: { from: loc.pathname + loc.search } });
    setBusy(true); setMsg('');
    try {
      await api('/bookings', { method: 'POST', body: { listingId: l.id, startDate: start, endDate: end } });
      nav('/trips', { state: { toast: 'Reservation confirmed. Have a great stay!' } });
    } catch (e) { setMsg(e.message); setBusy(false); }

function Month({ date, start, end, hover, setHover, onPick, todayISO, className = '' }) {
  const y = date.getFullYear();
  const m = date.getMonth();
  const offset = new Date(y, m, 1).getDay();
  const count = new Date(y, m + 1, 0).getDate();
  const cells = [...Array(offset).fill(0), ...Array.from({ length: count }, (_, i) => i + 1)];
  const tail = end || (start && hover > start ? hover : '');

  return (
    <div className={className}>
      <p className="mb-4 text-center text-[15px] font-semibold">
        {date.toLocaleDateString('en-IN', { month: 'long', year: 'numeric' })}
      </p>
      <div className="mb-1 grid grid-cols-7 text-center text-xs font-medium text-[#717171]">
        {['S', 'M', 'T', 'W', 'T', 'F', 'S'].map((d, i) => <span key={i} className="py-2">{d}</span>)}
      </div>
      <div className="grid grid-cols-7" onMouseLeave={() => setHover('')}>
        {cells.map((d, i) => {
          if (!d) return <span key={i} />;
          const id = iso(new Date(y, m, d));
          const past = id < todayISO;
          const selected = id === start || id === end;
          const mid = start && tail && id > start && id < tail;
          const bandRight = id === start && tail;
          const bandLeft = start && tail && id === tail && tail !== start;
          return (
            <div key={i} className="relative flex h-12 items-center justify-center">
              {mid && <span className="absolute inset-0 bg-[#F7F7F7]" />}
              {bandRight && <span className="absolute inset-y-0 right-0 w-1/2 bg-[#F7F7F7]" />}
              {bandLeft && <span className="absolute inset-y-0 left-0 w-1/2 bg-[#F7F7F7]" />}
              <button
                type="button"
                disabled={past}
                onMouseEnter={() => setHover(id)}
                onClick={() => onPick(id)}
                aria-label={id}
                className={`relative z-10 h-12 w-12 rounded-full border border-transparent text-sm font-medium ${
                  past
                    ? 'text-[#B0B0B0] line-through'
                    : selected
                    ? 'bg-[#222222] text-white'
                    : 'text-[#222222] hover:border-[#222222]'
                }`}
              >
                {d}
              </button>
            </div>
          );
        })}
      </div>
    </div>
  );
}

import { createContext, useCallback, useContext, useEffect, useMemo, useState } from 'react';
import { useLocation, useNavigate } from 'react-router-dom';
import { api } from './api';
import { useAuth } from './AuthContext';

const Ctx = createContext({
  items: [],
  ids: new Set(),
  loading: false,
  isSaved: () => false,
  toggle: async () => false,
  reload: async () => {},
});

export const useWishlist = () => useContext(Ctx);

export function WishlistProvider({ children }) {
  const { user } = useAuth();
  const navigate = useNavigate();
  const loc = useLocation();
  const [items, setItems] = useState([]);
  const [loading, setLoading] = useState(false);

  const reload = useCallback(async () => {
    if (!user) {
      setItems([]);
      return;
    }
    setLoading(true);
    try {
      setItems(await api('/wishlists'));
    } catch {
      /* keep the previous list */
    } finally {
      setLoading(false);
    }
  }, [user]);

  useEffect(() => {
    reload();
  }, [reload]);

  const ids = useMemo(() => new Set(items.map((i) => i.id)), [items]);

  // toggle(listing) or toggle(listingId). Pass the listing object for the smoothest optimistic update.
  const toggle = useCallback(
    async (target) => {
      if (!user) {
        navigate('/login', { state: { from: loc.pathname } });
        return false;
      }
      const id = typeof target === 'object' ? target.id : target;
      const wasSaved = ids.has(id);
      setItems((prev) => (wasSaved ? prev.filter((i) => i.id !== id) : typeof target === 'object' ? [target, ...prev] : prev));
      try {
        const res = await api('/wishlists', { method: 'POST', body: { listingId: id } });
        if (typeof target !== 'object') await reload();
        return res.saved;
      } catch {
        await reload(); // undo the optimistic change
        return wasSaved;
      }
    },
    [user, ids, navigate, loc.pathname, reload]
  );

  const value = useMemo(() => ({ items, ids, loading, isSaved: (id) => ids.has(id), toggle, reload }), [items, ids, loading, toggle, reload]);
  return <Ctx.Provider value={value}>{children}</Ctx.Provider>;
}

const iconCache = {};
function pinIcon(price, active) {
  const key = `${price}-${active}`;
  if (!iconCache[key]) {
    iconCache[key] = L.divIcon({
      className: 'price-pin-wrap',
      html: `<div class="price-pin${active ? ' active' : ''}">₹${Number(price).toLocaleString('en-IN')}</div>`,
      iconSize: [0, 0],
    });
  }
  return iconCache[key];
}

const hasCoords = (l) => Number.isFinite(l.latitude) && Number.isFinite(l.longitude);

function FitBounds({ points }) {
  const map = useMap();
  useEffect(() => {
    if (!points.length) return;
    map.fitBounds(L.latLngBounds(points.map((p) => [p.latitude, p.longitude])), { padding: [60, 60], maxZoom: 15 });
  }, [points, map]);
  return null;
}

export default function MapView({ listings = [] }) {
  const [activeId, setActiveId] = useState(null);
  const points = useMemo(() => listings.filter(hasCoords), [listings]);

  if (!points.length) {

      <div className="flex h-[60vh] items-center justify-center rounded-2xl border border-dashed border-neutral-300 text-center text-sm text-neutral-500">
        None of these places have a map location yet.
        <br />
        Import the dataset, or switch back to the list.
      </div>
    );
  }

  return (
    <div className="relative isolate h-[72vh] min-h-[420px] overflow-hidden rounded-2xl border border-neutral-200">
      <style>{css}</style>
      <MapContainer center={[points[0].latitude, points[0].longitude]} zoom={12} scrollWheelZoom style={{ height: '100%', width: '100%' }}>

        <FitBounds points={points} />
        {points.map((l) => (
          <Marker
            key={l.id}
            position={[l.latitude, l.longitude]}

            zIndexOffset={activeId === l.id ? 1000 : 0}
